<script>
function checkWater() {
let ph = parseFloat(document.getElementById("ph").value);
let turbidity = parseFloat(document.getElementById("turbidity").value);

let result = document.getElementById("result");

if ((ph >= 6.5 && ph <= 8.5) && turbidity <= 5) {
result.innerHTML = "✅ Safe Water";
result.style.color = "green";
}
else if ((ph >= 5.5 && ph <= 9) && turbidity <= 10) {
result.innerHTML = "⚠ Moderately Polluted";
result.style.color = "orange";
}
else {
result.innerHTML = "❌ Highly Polluted";
result.style.color = "red";
}
}
</script>
